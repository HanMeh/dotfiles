  "Build and run Rust code, allowing user input."
  (interactive)
  (cargo-process-run)
  (let* ((orig-win (selected-window))
         (run-win (display-buffer (get-buffer "*Cargo Run*") nil 'visible)))
    (select-window run-win)
    (comint-mode)
    (read-only-mode 0)
    (select-window orig-win)))

;; 2. Loop through the registry list and automatically install missing grammars
(defun my/ensure-all-treesit-grammars ()
  "Loop through the configured language list and compile missing binaries."
  (interactive)
  (when (and (fboundp 'treesit-available-p)
             (treesit-available-p))
    (dolist (mapping treesit-language-source-alist)
      (let ((lang (car mapping)))
        (unless (treesit-language-available-p lang)
          (message "Tree-sitter: Automatically installing '%s' grammar..." lang)
          (treesit-install-language-grammar lang))))))

;; 3. Run the automated installer check safely on startup
(add-hook 'after-init-hook #'my/ensure-all-treesit-grammars)

(defun my/bazel-test-current-file ()
  "Asynchronously run the closest Bazel test target for the current file."
  (interactive)
  (when (buffer-file-name)
    (let* ((file-path (buffer-file-name))
           ;; Smart Target Finder: Ask bazel query what test target claims this file.
           ;; If query fails or is slow, fallback to a path-based guess.
           (bazel-target 
            (or (let ((query-cmd (format "bazel query 'attr(srcs, %s, kind(test, //...))' 2>/dev/null" 
                                         (file-name-nondirectory file-path))))
                  (string-trim (shell-command-to-string query-cmd)))
                "//..."))) ;; Worst case fallback: run everything or adjust to a default
      
      ;; If a target is found, run it through the Emacs compilation buffer
      (if (and bazel-target (not (string-empty-p bazel-target)))
          (progn
            (message "Bazel: Testing target %s..." bazel-target)
            ;; compile-command tells 'compile' exactly what to execute
            (setq-local compile-command (format "bazel test %s" bazel-target))
            ;; Runs natively, allowing you to use 'C-x `' to jump to errors
            (compilation-start compile-command))
        (message "Bazel: No matching test target found for this file.")))))

;; Define a minor mode or function to easily toggle this behavior per buffer
(define-minor-mode my/bazel-autosave-test-mode
  "Minor mode to automatically trigger Bazel tests on buffer save."
  :lighter " BzTest"
  (if my/bazel-autosave-test-mode
      (add-hook 'after-save-hook #'my/bazel-test-current-file nil t)
    (remove-hook 'after-save-hook #'my/bazel-test-current-file t)))
